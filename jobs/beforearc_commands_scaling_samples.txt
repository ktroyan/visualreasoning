*** All commands to run the scaling samples experiments in COGITAO ***

NOTE:
- We specify the relevant parameters based on the default configs.
- We changed from bash ./jobs/run_experiment_nohup.sh to sbatch jobs/run_experiment_wrapper.slurm since the new cluster uses SLURM.
- The arg --gpu_id 0 will be ignored under SLURM; SLURM sets CUDA_VISIBLE_DEVICES for us.


### 100k samples, 100 epochs
# Models
sbatch jobs/run_experiment_wrapper.slurm --sweep_enabled true --sweep_config "configs/sweeps/beforearc/comp_scaling_samples_100k_100epochs.yaml" --backbone resnet --visual_tokens_enabled false --add_log_name "ResNet"
sbatch jobs/run_experiment_wrapper.slurm --sweep_enabled true --sweep_config "configs/sweeps/beforearc/comp_scaling_samples_100k_100epochs.yaml" --backbone vit --visual_tokens_enabled false --ape_enabled true --ape_type learn --ape_mixer sum --ope_enabled false --rpe_enabled false --add_log_name "Vanilla-TF"
sbatch jobs/run_experiment_wrapper.slurm --sweep_enabled true --sweep_config "configs/sweeps/beforearc/comp_scaling_samples_100k_100epochs.yaml" --backbone vit --visual_tokens_enabled true --ape_enabled true --ape_type 2dsincos --ape_mixer weighted_sum_vec --ope_enabled true --rpe_enabled true --rpe_type rope --num_reg_tokens 6 --add_log_name "Grid-TF"
sbatch jobs/run_experiment_wrapper.slurm --sweep_enabled true --sweep_config "configs/sweeps/beforearc/comp_scaling_samples_100k_100epochs.yaml" --backbone looped_vit --visual_tokens_enabled true --ape_enabled true --ape_type 2dsincos --ape_mixer sum --ope_enabled false --rpe_enabled true --rpe_type rope --add_log_name "PL-TF"
